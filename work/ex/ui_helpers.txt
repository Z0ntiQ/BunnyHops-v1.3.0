pub fn input_value(ev: web_sys::Event) -> String {
    ev.target()
        .unwrap()
        .dyn_into::<web_sys::HtmlElement>()
        .unwrap()
        .dyn_ref::<web_sys::HtmlInputElement>()
        .map(|i| i.value())
        .or_else(|| {
            ev.target()
                .unwrap()
                .dyn_ref::<web_sys::HtmlTextAreaElement>()
                .map(|t| t.value())
        })
        .unwrap_or_default()
}

pub fn input_checked(ev: web_sys::Event) -> bool {
    ev.target()
        .unwrap()
        .dyn_ref::<web_sys::HtmlInputElement>()
        .map(|i| i.checked())
        .unwrap_or(false)
}

pub fn select_value(ev: web_sys::Event) -> String {
    ev.target()
        .unwrap()
        .dyn_into::<web_sys::HtmlSelectElement>()
        .map(|s| s.value())
        .unwrap_or_default()
}

pub fn new_id() -> String {
    let uuid = web_sys::window()
        .and_then(|w| w.crypto().ok())
        .map(|c| c.random_uuid())
        .unwrap_or_default();
    if !uuid.is_empty() {
        return uuid;
    }
    format!("{}-{}", js_now(), js_random())
}

pub fn js_now() -> f64 {
    web_sys::window().map(|w| w.performance().unwrap().now()).unwrap_or(0.0)
}

pub fn js_random() -> f64 {
    js_sys::Math::random()
}
